How to Use UltSQL

Simple, straightforward steps to integrate UltSQL into Dart or Flutter applications.

Dart Code Integration 
import 'package:hybrid_sql_engine/engine/executor/interpreter.dart';
import 'package:hybrid_sql_engine/engine/executor/value.dart';

void main() async {
  // 1. Initialize Database (Switchable: ':memory:' or '/path/to/db')
  final db = Database(':memory:');
  await db.init();

  final interpreter = Interpreter(db);

  // 2. Execute SQL DDL / DML
  await interpreter.executeScript('''
    CREATE TABLE employees (id INT PRIMARY KEY, name TEXT, salary DOUBLE);
    INSERT INTO employees VALUES (1, 'Alice Vance', 125000.0);
    INSERT INTO employees VALUES (2, 'Bob Builder', 98000.0);
  ''');

  // 3. Prepared Statement Batch Execution
  final prep = db.prepare("INSERT INTO employees VALUES (?, ?, ?);");
  prep.executeBatchSync([
    [DbInt(3), DbText('Charlie'), DbDouble(140000.0)],
    [DbInt(4), DbText('Diana'), DbDouble(92000.0)],
  ]);

  // 4. Query Results
  final result = await interpreter.executeScript("SELECT * FROM employees WHERE salary > 100000.0;");
  for (final row in result.rows) {
    print("Employee: \${row[1].value}, Salary: \${row[2].value}");
  }

  await db.close();
}
